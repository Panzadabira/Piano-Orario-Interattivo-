        function initDragAndDrop() {
            const draggables = document.querySelectorAll('.draggable');
            const dropZones = document.querySelectorAll('.drop-zone');

            draggables.forEach(draggable => {
                draggable.removeEventListener('dragstart', dragStart);
                draggable.removeEventListener('dragend', dragEnd);
                draggable.addEventListener('dragstart', dragStart);
                draggable.addEventListener('dragend', dragEnd);
            });

            dropZones.forEach(zone => {
                zone.removeEventListener('dragover', dragOver);
                zone.removeEventListener('dragleave', dragLeave);
                zone.removeEventListener('drop', drop);
                zone.addEventListener('dragover', dragOver);
                zone.addEventListener('dragleave', dragLeave);
                zone.addEventListener('drop', drop);
            });
        }

        function dragStart(e) {
            e.dataTransfer.setData('text/plain', e.target.id);
            setTimeout(() => e.target.classList.add('opacity-40'), 0);
        }

        function dragEnd(e) {
            e.target.classList.remove('opacity-40');
            document.querySelectorAll('.drop-zone').forEach(z => z.classList.remove('drag-over'));
            checkBacklogEmpty();
        }

        function dragOver(e) {
            e.preventDefault(); 
            const zone = e.target.closest('.drop-zone');
            if (zone) zone.classList.add('drag-over');
        }

        function dragLeave(e) {
            const zone = e.target.closest('.drop-zone');
            if (zone) zone.classList.remove('drag-over');
        }

        function drop(e) {
            e.preventDefault();
            const zone = e.target.closest('.drop-zone');
            if (zone) {
                zone.classList.remove('drag-over');
                const id = e.dataTransfer.getData('text/plain');
                const draggableElement = document.getElementById(id);
                if (draggableElement) {
                    zone.appendChild(draggableElement);
                }
            }
        }

        function moveCol(btn, dir) {
            const th = btn.closest('th');
            const table = document.getElementById('mainTable');
            const currentIndex = th.cellIndex;
            const targetIndex = currentIndex + dir;
            const theadTr = th.parentNode;
            
            if (targetIndex < 1 || targetIndex >= theadTr.children.length) return;

            if (dir === 1) {
                theadTr.insertBefore(th, theadTr.children[targetIndex + 1]);
            } else {
                theadTr.insertBefore(th, theadTr.children[targetIndex]);
            }

            const tbodyTrs = table.querySelectorAll('tbody tr');
            tbodyTrs.forEach(tr => {
                const td = tr.children[currentIndex];
                if (dir === 1) {
                    tr.insertBefore(td, tr.children[targetIndex + 1]);
                } else {
                    tr.insertBefore(td, tr.children[targetIndex]);
                }
            });
        }

        function exportData() {
            const data = { columns: [], rows: [], tasks: [] };

            // 1. Leggi Colonne
            const ths = document.querySelectorAll('#mainTable thead th');
            for (let i = 1; i < ths.length; i++) { // Salta colonna 0 (Categoria)
                data.columns.push({
                    name: ths[i].querySelector('.col-name').innerText,
                    sub: ths[i].querySelector('.col-sub').innerText
                });
            }

            // 2. Leggi Righe e Task
            const trs = document.querySelectorAll('#mainTable tbody tr');
            trs.forEach((tr, rowIndex) => {
                const tdCat = tr.querySelector('td:first-child');
                // Estrae il colore base dalla classe (es: bg-blue-50/50 -> blue)
                let rowColor = 'gray';
                const match = tdCat.className.match(/bg-([a-z]+)-50/);
                if (match) rowColor = match[1];

                data.rows.push({
                    name: tdCat.innerText,
                    color: rowColor
                });

                const tds = tr.querySelectorAll('td');
                for (let colIndex = 1; colIndex < tds.length; colIndex++) {
                    const dropZone = tds[colIndex].querySelector('.drop-zone');
                    if (dropZone) {
                        const tasks = dropZone.querySelectorAll('.draggable');
                        tasks.forEach(task => {
                            data.tasks.push({
                                id: task.id,
                                name: task.querySelector('.task-title').innerText,
                                sub: task.querySelector('.task-subtext') ? task.querySelector('.task-subtext').innerText : '',
                                color: task.getAttribute('data-color'),
                                location: { type: 'grid', rowIndex: rowIndex, colIndex: colIndex - 1 }
                            });
                        });
                    }
                }
            });

            // 3. Leggi Backlog Task
            const backlogTasks = document.querySelectorAll('#backlog .draggable');
            backlogTasks.forEach(task => {
                data.tasks.push({
                    id: task.id,
                    name: task.querySelector('.task-title').innerText,
                    sub: task.querySelector('.task-subtext') ? task.querySelector('.task-subtext').innerText : '',
                    color: task.getAttribute('data-color'),
                    location: { type: 'backlog' }
                });
            });

            // Genera il file e forza il download
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(data, null, 2));
            const downloadAnchorNode = document.createElement('a');
            downloadAnchorNode.setAttribute("href", dataStr);
            downloadAnchorNode.setAttribute("download", "piano_orario.json");
            document.body.appendChild(downloadAnchorNode); 
            downloadAnchorNode.click();
            downloadAnchorNode.remove();
        }

        function importData(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = JSON.parse(e.target.result);
                    rebuildTableFromData(data);
                    document.getElementById('importFile').value = ''; // Reset file input
                } catch (error) {
                    alert("Si è verificato un errore durante la lettura del file JSON. Assicurati che sia il file corretto.");
                    console.error(error);
                }
            };
            reader.readAsText(file);
        }

        function rebuildTableFromData(data) {

            theadTr.innerHTML = '<th class="p-4 font-semibold w-1/6 border-r border-gray-200 bg-gray-100 sticky left-0 z-20">Categoria</th>';
            
            data.columns.forEach(col => {
                const th = document.createElement('th');
                th.className = "p-4 font-semibold text-center border-r border-gray-200 relative group min-w-[120px]";
                th.innerHTML = `
                    <div class="flex flex-col items-center justify-center">
                        <span class="col-name text-gray-700">${col.name}</span>
                        <span class="col-sub text-xs font-normal text-gray-500 mt-1">${col.sub}</span>
                    </div>
                    ${getColControlsHTML()}
                `;
                theadTr.appendChild(th);
            });

            // 2. Ricostruisci Body (Righe)
            const tbody = document.querySelector('#mainTable tbody');
            tbody.innerHTML = ''; 
            data.rows.forEach(row => {
                const tr = document.createElement('tr');
                tr.className = "border-b border-gray-200 transition duration-150";
                
                const tdCat = document.createElement('td');
                tdCat.className = `p-4 font-semibold text-${row.color}-700 bg-${row.color}-50/50 border-r border-gray-200 sticky left-0 z-10`;
                tdCat.innerText = row.name;
                tr.appendChild(tdCat);

                for(let i = 0; i < data.columns.length; i++) {
                    const td = document.createElement('td');
                    td.className = "border-r border-gray-200 align-top p-2";
                    td.innerHTML = '<div class="drop-zone flex flex-col gap-2 h-full min-h-[80px]"></div>';
                    tr.appendChild(td);
                }
                tbody.appendChild(tr);
            });

            // Pulisci il backlog attuale
            const backlog = document.getElementById('backlog');
            backlog.querySelectorAll('.draggable').forEach(el => el.remove());

            // 3. Riposiziona i Task
            taskCounter = 100; // Reset counter base

            data.tasks.forEach(t => {
                // Aggiorna il counter se l'id del task importato è più alto
                const idNum = parseInt(t.id.replace('task-', ''));
                if (!isNaN(idNum) && idNum >= taskCounter) {
                    taskCounter = idNum;
                }

                const borderClass = `border-${t.color === 'gray' ? 'gray-300' : t.color + '-200'}`;
                const borderLeftClass = `border-l-4 border-l-${t.color === 'gray' ? 'gray-500' : t.color + '-500'}`;
                
                const taskDiv = document.createElement('div');
                taskDiv.id = t.id;
                taskDiv.draggable = true;
                taskDiv.className = `draggable bg-white border ${borderClass} p-3 rounded shadow-sm ${borderLeftClass} relative group min-w-[150px]`;
                taskDiv.setAttribute('data-color', t.color);
                taskDiv.innerHTML = getTaskHTML(t.id, t.name, t.sub, t.color);

                if (t.location.type === 'backlog') {
                    backlog.appendChild(taskDiv);
                } else if (t.location.type === 'grid') {
                    const targetTr = tbody.children[t.location.rowIndex];
                    if (targetTr) {
                        const targetTd = targetTr.children[t.location.colIndex + 1]; // +1 per skippare la colonna Categoria
                        if (targetTd) {
                            const dropZone = targetTd.querySelector('.drop-zone');
                            if (dropZone) dropZone.appendChild(taskDiv);
                        }
                    }
                }
            });

            initDragAndDrop();
            checkBacklogEmpty();
        }

        // --- Utility ---
        function checkBacklogEmpty() {
            const backlog = document.getElementById('backlog');
            const text = document.getElementById('backlog-empty-text');
            const hasDraggables = backlog.querySelectorAll('.draggable').length > 0;
            if (hasDraggables) {
                text.style.display = 'none';
            } else {
                text.style.display = 'block';
            }
        }

        document.addEventListener('DOMContentLoaded', () => {
            initDragAndDrop();
            checkBacklogEmpty();
        });

    </script>
